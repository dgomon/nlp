    \section*{Question 4}
    \subsection*{a}
    Defining the cost of choosing a word $w_i$ as $c(w_i) =
        -\log P(w_i) = \log \frac{1}{P(w_i)}$, where $c(w_i) \ge 0$ since $0 < P(w_i) \le 1$,
        the problem is equivalent to minimizing the total cost $\sum_{i=1}^k c(w_i)$:
    \begin{align*}
        \argmax_{w_1, \dots, w_k} \prod_{i=1}^k P(w_i)
        &= \argmax_{w_1, \dots, w_k} (\log \prod_{i = 1}^{k}P(w_i))
            && \text{($a > b \iff \log a > \log b$)} \\
        &= \argmax_{w_1, \dots, w_k} \sum_{i=1}^k \log P(w_i)
            && (\log(ab) = \log(a) + \log(b)) \\
        &= \argmin_{w_1, \dots, w_k} \sum_{i=1}^k -\log P(w_i) \\
        &= \argmin_{w_1, \dots, w_k} \sum_{i=1}^k \log \frac{1}{P(w_i)}
            && (-\log(a) = \log(\frac{1}{a})) \\
        &= \argmin_{w_1, \dots, w_k} \sum_{i=1}^k c(w_i)
    \end{align*}

    \subsection*{b}
    \begin{tabbing}
    \\
    \textbf{function} \textsc{Min-Cost}(string $S$, vocabulary $V$) \textbf{returns} \textit{min-cost} \\
    \quad \# \textit{Initialization} \\
    1. \quad $length \leftarrow \textsc{Length}(S)$ \\
    2. \quad Create a cost vector $A[length + 1]$ \\
       \quad\quad \textit{$\# A[j]$ contains minimum cost of segmenting the prefix of length $j$} \\
    3. \quad $A[0] \leftarrow 0$ \\
    4. \quad $A[1 .. length] \leftarrow \infty$ \\
    \\
    5. \quad \textbf{for} each starting word index $i$ \textbf{from} 0 \textbf{to} $length - 1$ \textbf{do} \\
    6. \quad\quad \textbf{if} $A[i] \neq \infty$ \\
       \quad\quad\quad \# \textit{$i$-th letter is reachable from the beginning of the word} \\
    7. \quad\quad\quad \textbf{for} each ending word index $j$ \textbf{from} i + 1 \textbf{to} $length$ \textbf{do} \\
    8. \quad\quad\quad\quad \textbf{if} $S[i + 1 \dots j] \in V$ \\
       \quad\quad\quad\quad\quad \# \textit{Choose the cheaper path among the two} \\
    9. \quad\quad\quad\quad\quad $A[j] \leftarrow \min(A[j], A[i] - \log(P(S[i + 1 \dots j])))$ \\
    10. \quad\quad\quad\textbf{end if} \\
    11. \quad \textbf{end if}
    \\
    \\
    \quad \# \textit{Termination} \\
    12. \quad \textbf{return} $A[length]$
    \end{tabbing}

    \subsection*{c}
    We keep an additional array \texttt{B} of size $n + 1$.
    We can construct B by replacing step 9. with the following steps:
    \begin{tabbing}
        9.1 \quad \textbf{if} $S[i+1 \dots j] \in V$ and $(A[i] - \log(P((S[i + 1 \dots j]))) < A[j])$: \\
        \quad\quad\quad\quad \# \textit{Keep the minimal cost} \\
        9.2 \quad\quad $A[j] \leftarrow A[i] - \log(P((S[i + 1 \dots j])))$ \\
        \quad\quad\quad\quad \# \textit{Save starting index of the word ending at index }j \\
        9.3 \quad\quad $B[j] \leftarrow i + 1$ \\
        9.4 \quad \textbf{end if}
    \end{tabbing}
    Once B has been constructed, starting from j = length, B[j] will contain the start index of the word that ends in index j.
    Backtracking from the last word, we thus can collect all the sub-words that minimize the cost.
    Algorithm to get the optimal division thus is:\\
    \\
    \textbf{function} \textsc{Get-Optimal-Division}(string $S$, indices $B$) \textbf{returns} division W
    \begin{tabbing}
        \quad \# \textit{Initialization} \\
        1. \quad $W \leftarrow []$ \textit{\# Create an empty words vector } \\
        2. \quad $k \leftarrow length$ \textit {\# Create an index pointer}  \\
        \\
        3. \quad While $k > 0$: \\
        4. \quad\quad Let $start \leftarrow B[k]$. \\
        5. \quad\quad Prepend the substring $S[start \dots k]$ to $W$. \\
        6. \quad\quad Update $k \leftarrow start - 1$. \\
        \\
        \quad \# \textit{Termination} \\
        7. \quad \textbf{Return} $W$
    \end{tabbing}

    \subsection*{d}
    \subsubsection*{Time Complexity}
    Outer loop (index $i$) in Step 5. runs for n iterations.
    Inner loop (index $j$) in Step 7. runs for $i$ iterations, for a total of $\sum_{i = 1}^{n}\sum_{j = 1}^{i} \frac{n(n+1)}{2}$ iterations.
    Since it's given that complexity of lookup (step 8) is $O(1)$ and also assuming the complexity of assignment and
    log calculation operations in step 9, to be $O(1)$ as well, we obtain that the overall algorithm time complexity is $O(n^2)$.

    \subsubsection*{Space Complexity}
    Length of both $A$ and $B$ is $length$, and only $O(1)$ additional space gets allocated, thus space complexity is $\Theta(n)$

    \subsection*{e}
    Both inner and outer loops contain $L$ iterations, and given $L \ll n$, we assume exists constant $C$ such that $L < C$,
    thus reducing runtime time complexity to $O(1)$.
