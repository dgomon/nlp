    \section*{Question 4}
    \subsection*{a}
    Defining the cost of choosing a word $w_i$ as $c(w_i) =
        -\log P(w_i) = \log \frac{1}{P(w_i)}$, where $c(w_i) \ge 0$ since $0 < P(w_i) \le 1$,
        the problem is equivalent to minimizing the total cost $\sum_{i=1}^k c(w_i)$:
    \begin{align*}
        \argmax_{w_1, \dots, w_k} \prod_{i=1}^k P(w_i)
        &= \argmax_{w_1, \dots, w_k} (\log \prod_{i = 1}^{k}P(w_i))
            && \text{($a > b \iff \log a > \log b$)} \\
        &= \argmax_{w_1, \dots, w_k} \sum_{i=1}^k \log P(w_i)
            && (\log(ab) = \log(a) + \log(b)) \\
        &= \argmin_{w_1, \dots, w_k} \sum_{i=1}^k -\log P(w_i) \\
        &= \argmin_{w_1, \dots, w_k} \sum_{i=1}^k \log \frac{1}{P(w_i)}
            && (-\log(a) = \log(\frac{1}{a})) \\
        &= \argmin_{w_1, \dots, w_k} \sum_{i=1}^k c(w_i)
    \end{align*}

    \quad Create a cost vector $A[length]$ \\
    \\
    \quad \# \textit{Initialization: distance to zeroth letter is zero, distance to other letters is infinity} \\
    \quad $A[0] = 0$ \\
    \quad $A[1 .. length] = \infty$ \\
    \\
    \quad \textbf{for} each starting word index $i$ \textbf{from} 0 \textbf{to (not including)} $length$ \textbf{do} \\
    \quad\quad \textbf{if} $A[i] == \infty$ \\
    \quad\quad\quad \# \textit{letter $i$ is unreachable from beginning of the word} \\
    \quad\quad\quad\textbf{continue}  \\
    \qquad \textbf{for} each ending word index $j$ \textbf{from} i + 1 \textbf{to (not including)} $length$ \textbf{do} \\
    \quad\quad\quad w $\leftarrow$ \textbf{Get-Weight}(S, V, i, j) \\
    \quad\quad\quad A[j] = min(A[j], A[i] + weight(i, j))
    \\
    \\
    \quad \# \textit{Termination} \\
    \quad \textbf{return} $A[length - 1]$
